# Environment Variables Template
# Copy this to .env file in the Solution1 directory
# Docker Compose will automatically load .env files
#
# IMPORTANT: Never commit .env files to git (they contain sensitive passwords!)

# SQL Server Configuration
# Change this to a strong password in production!
SA_PASSWORD=YourStrong@Password123!

# JWT Secret Key
# Change this to a secure random string in production!
# Minimum 32 characters recommended for security
JWT_SECRET=SliacFantasyFootball2025SecureJwtKeyForAuthenticationAndAuthorization!@#MinimumLengthRequired
